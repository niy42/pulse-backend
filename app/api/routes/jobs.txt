from fastapi import APIRouter, HTTPException
from uuid import uuid4
from datetime import datetime
import asyncio
import re

from pydantic import BaseModel, field_validator
from typing import Any, List

from app.services.process_video import process_video
from app.services.user_service import get_or_create_user

router = APIRouter()

# 🔥 In-memory store
jobs = {}


# -------------------------
# 📦 Schemas
# -------------------------
class CreateChatRequest(BaseModel):
    user_id: str


class SendMessageRequest(BaseModel):
    content: str

    @field_validator("content")
    def validate_url(cls, v):
        if not re.match(r"(https?://)?(www\.)?(youtube\.com|youtu\.be)/.+", v):
            raise ValueError("Invalid YouTube URL")
        return v


class Message(BaseModel):
    role: str  # "user" | "assistant"
    content: Any


# -------------------------
# 🚀 Create New Chat
# -------------------------
@router.post("/chats")
async def create_chat(data: CreateChatRequest):
    job_id = str(uuid4())

    jobs[job_id] = {
        "id": job_id,
        "user_id": data.user_id,
        "title": "New Chat",
        "messages": [],
        "created_at": datetime.utcnow().isoformat(),
    }

    return jobs[job_id]


# -------------------------
# 💬 Send Message (Start Processing)
# -------------------------
@router.post("/chats/{chat_id}/messages")
async def send_message(chat_id: str, data: SendMessageRequest):
    job = jobs.get(chat_id)

    if not job:
        raise HTTPException(status_code=404, detail="Chat not found")

    user = get_or_create_user(job["user_id"])

    # 🚫 enforce free plan
    if user["plan"] == "free" and user["requests_used"] >= 3:
        raise HTTPException(status_code=403, detail="Upgrade to Pro to continue")

    user["requests_used"] += 1

    # 👤 Add user message
    job["messages"].append(
        {
            "role": "user",
            "content": data.content,
        }
    )

    # ⚙️ Async processing
    asyncio.create_task(run_job(chat_id, data.content))

    return {"status": "processing"}


# -------------------------
# 📚 Get All Chats (Sidebar)
# -------------------------
@router.get("/chats")
async def get_chats(user_id: str):
    user_chats = [job for job in jobs.values() if job["user_id"] == user_id]

    # newest first
    return user_chats[::-1]


# -------------------------
# 💬 Get Single Chat
# -------------------------
@router.get("/chats/{chat_id}")
async def get_chat(chat_id: str):
    job = jobs.get(chat_id)

    if not job:
        raise HTTPException(status_code=404, detail="Chat not found")

    return job


# -------------------------
# ⚙️ Background Processor
# -------------------------
async def run_job(chat_id: str, url: str):
    job = jobs[chat_id]

    try:
        result = await process_video(url)

        # 🤖 Assistant response
        job["messages"].append(
            {
                "role": "assistant",
                "content": {
                    "hooks": result.hooks,
                    "insights": result.insights,
                    "contrarian": result.contrarian,
                    "summary": result.summary,
                    "quotes": result.quotes,
                },
            }
        )

        # 🧠 Auto-title chat
        if job["title"] == "New Chat":
            job["title"] = url[:50]

    except Exception as e:
        job["messages"].append({"role": "assistant", "content": {"error": str(e)}})
